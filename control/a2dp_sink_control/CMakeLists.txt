cmake_minimum_required(VERSION 3.22)

if(NOT CONTROL_VARIANT MATCHES "^[1-9]$")
    message(FATAL_ERROR "Select exactly one control variant (1..9) through PlatformIO")
endif()
if(CONTROL_VARIANT STREQUAL "2")
    set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults;${CMAKE_CURRENT_LIST_DIR}/sdkconfig.main-config.defaults")
endif()
if(CONTROL_VARIANT STREQUAL "9")
    if(CONTROL_OFFICIAL_ENV STREQUAL "control-official-i2s-reference")
        set(official_defaults "sdkconfig.official-160-sleep.defaults")
    elseif(CONTROL_OFFICIAL_ENV MATCHES "^control-official-(160-nosleep|240-sleep|240-nosleep)$")
        string(REPLACE "control-" "sdkconfig." official_defaults "${CONTROL_OFFICIAL_ENV}.defaults")
    else()
        message(FATAL_ERROR "Select one official CPU/sleep environment through PlatformIO")
    endif()
    set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults;${CMAKE_CURRENT_LIST_DIR}/${official_defaults}")
endif()

include($ENV{IDF_PATH}/tools/cmake/project.cmake)
idf_build_set_property(CONTROL_VARIANT "${CONTROL_VARIANT}")
if(CONTROL_VARIANT STREQUAL "9")
    idf_build_set_property(CONTROL_OFFICIAL_ENV "${CONTROL_OFFICIAL_ENV}")
endif()
idf_build_set_property(MINIMAL_BUILD ON)
project(a2dp_sink_control)
